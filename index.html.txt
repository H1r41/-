    const GAS_API_URL = 'https://script.google.com/macros/s/AKfycbziVeevNjJNQGOF0zBg8xGnVmf_9RbcycpaPAb_bKU77krjisBFQvjNu70wACcbUe5mog/exec';

    let currentMode = 'quick';

    let allAccounts = [];
    let ledgerOptions = {};

    let selectedMulti = { genres: [], memos: [] };

    function getStoredSecret() {
      return localStorage.getItem('kakeibo_app_secret') || '';
    }

    async function callGas(action, payload) {
      if (!GAS_API_URL || GAS_API_URL.indexOf('https://script.google.com/') !== 0) {
        throw new Error('index.html の GAS_API_URL を設定してください');
      }
      const res = await fetch(GAS_API_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({
          secret: getStoredSecret(),
          action: action,
          payload: payload || {}
        })
      });
      const json = await res.json();
      if (!json.ok) {
        if (json.error === '合言葉が一致しません') {
          localStorage.removeItem('kakeibo_app_secret');
          showLockScreen('合言葉が違います。もう一度入力してください。');
        }
        throw new Error(json.error || '通信エラーが発生しました');
      }
      return json.data;
    }

    function showLockScreen(msg) {
      document.getElementById('authCard').classList.remove('hidden');
      document.getElementById('mainApp').classList.add('hidden');
      document.getElementById('footerBar').classList.add('hidden');
      document.getElementById('authError').textContent = msg || '';
    }

    async function saveSecretAndLoad() {
      const val = document.getElementById('secretInput').value.trim();
      if (!val) {
        document.getElementById('authError').textContent = '合言葉を入力してください';
        return;
      }
      localStorage.setItem('kakeibo_app_secret', val);
      document.getElementById('authError').textContent = '確認中...';
      await bootApp();
    }

    async function bootApp() {
      document.getElementById('authCard').classList.add('hidden');

      document.getElementById('footerBar').classList.remove('hidden');
      try {
        const data = await callGas('getInitialData');
        populateData(data);

      }
    }

    window.onload = async function() {
      const todayStr = formatDate(new Date());
      document.getElementById('qDate').value = todayStr;
      document.getElementById('ptDate').value = todayStr;

    function formatDate(d) {
      return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    }

    function getSubmitLabel(mode) {
      if (mode === 'quick') return currentInout === '支出' ? '支出を記録する' : '収入を記録する';
      if (mode === 'point') return '月末ポイントを記録する';
      if (mode === 'transfer') return '振替を記録する';
      if (mode === 'card') return '引落しを予約する';
      return '振込みを予約する';
    }

    function switchMode(mode) {
      currentMode = mode;
      document.getElementById('status').textContent = '';
      const map = { quick: 'tabQuick', point: 'tabPoint', transfer: 'tabTransfer', card: 'tabCard', payout: 'tabPayout' };
      const forms = { quick: 'quickForm', point: 'pointForm', transfer: 'transferForm', card: 'cardForm', payout: 'payoutForm' };

      Object.keys(map).forEach(k => {
        document.getElementById(map[k]).classList.toggle('active', k === mode);
        document.getElementById(forms[k]).classList.toggle('hidden', k !== mode);
      });

      const btn = document.getElementById('execBtn');
      btn.classList.remove('pt-mode', 'card-mode', 'payout-mode');
      if (mode === 'point') btn.classList.add('pt-mode');
      if (mode === 'card') { btn.classList.add('card-mode'); onCardChange(); }
      if (mode === 'payout') { btn.classList.add('payout-mode'); onPayoutChange(); }
      if (!btn.disabled) btn.textContent = getSubmitLabel(mode);
    }

    function populateData(data) {
      if (data && data.error) { showError(data.error); return; }
      allAccounts = data.accounts || [];
      ledgerOptions = data.ledgerOptions || {};
      allTags = data.allTags || [];

      const groupedAccHtml = buildGroupedAccountOptions(allAccounts);
      document.getElementById('qAccount').innerHTML = groupedAccHtml;
      document.getElementById('fromAccount').innerHTML = groupedAccHtml;
      document.getElementById('toAccount').innerHTML = groupedAccHtml;

      const pointAccounts = allAccounts.filter(a => a.accountType === 'ポイント');
      const ptHtml = pointAccounts.length
        ? pointAccounts.map(a => `<option value="${a.id}">${a.name}</option>`).join('')
        : '<option value="">ポイント口座なし</option>';
      document.getElementById('qSplitPointAccount').innerHTML = ptHtml;
      document.getElementById('ptAccountSelect').innerHTML = ptHtml;
      onPointAccountChange();

      document.getElementById('qTaxRate').innerHTML = (data.taxOptions || ['税込み'])
        .map(t => `<option value="${t}" ${t === '税込み' ? 'selected' : ''}>${t}</option>`).join('');

      document.getElementById('newTagColor').innerHTML = (data.colorOptions || [])
        .map(c => `<option value="${c.name}">${c.name}</option>`).join('');
      document.getElementById('newTagVisibility').innerHTML = (data.visibilityOptions || [])
        .map(v => `<option value="${v.name}">${v.name}</option>`).join('');

      refreshTagDropdowns();

      const cards = allAccounts.filter(a => a.accountType === 'クレジットカード');
      document.getElementById('cardSelect').innerHTML = cards.length
        ? cards.map(c => `<option value="${c.id}">${c.name}</option>`).join('')
        : '<option value="">クレジットカードなし</option>';

      const payouts = allAccounts.filter(a => a.accountType === '報酬・立替');
      document.getElementById('payoutSourceSelect').innerHTML = payouts.length
        ? payouts.map(p => `<option value="${p.id}">${p.name}</option>`).join('')
        : '<option value="">報酬・立替口座なし</option>';

      onCardChange();
      onPayoutChange();

      const btn = document.getElementById('execBtn');
      btn.disabled = false;
      btn.textContent = getSubmitLabel(currentMode);
    }

    function buildGroupedAccountOptions(accounts) {
      if (!accounts.length) return '<option value="">口座なし</option>';
      const order = ['現金・電子マネー', 'クレジットカード', '銀行', 'ポイント', '報酬・立替', 'その他'];
      const groups = {};
      accounts.forEach(a => {
        const key = order.includes(a.accountType) ? a.accountType : 'その他';
        if (!groups[key]) groups[key] = [];
        groups[key].push(a);
      });
      return order.filter(k => groups[k]).map(k =>
        `<optgroup label="${k}">${groups[k].map(a => `<option value="${a.id}">${a.name}</option>`).join('')}</optgroup>`
      ).join('');
    }

    function refreshTagDropdowns() {
      const renderOpts = (propName, placeholder) => {
        const list = ledgerOptions[propName] || [];
        return `<option value="">${placeholder}</option>` + list.map(o =>
          o.isSep ? `<option disabled>${o.name}</option>` : `<option value="${o.name}">${o.name}</option>`
        ).join('');
      };
      document.getElementById('qMainCat').innerHTML = renderOpts('大項目', '選択...');
      document.getElementById('qSubCat').innerHTML = renderOpts('小項目', '選択...');
      document.getElementById('qGenreSelect').innerHTML = renderOpts('界隈ジャンル', '追加...');
      document.getElementById('qMemoSelect').innerHTML = renderOpts('用途・メモ', '追加...');

      const subList = (ledgerOptions['小項目'] || []).filter(o => !o.isSep && o.name.includes('【振替】') && !o.name.includes('引き落とし'));
      document.getElementById('purpose').innerHTML = subList.length
        ? subList.map(o => `<option value="${o.name}">${o.name}</option>`).join('')
        : '<option value="">振替用小項目なし</option>';
    }

    function togglePointSplit() {
      const on = document.getElementById('qUsePointSplit').checked;
      document.getElementById('pointSplitFields').classList.toggle('hidden', !on);
      updateSplitPreview();
    }

    function updateSplitPreview() {
      if (!document.getElementById('qUsePointSplit').checked) return;
      const total = Number(document.getElementById('qAmount').value) || 0;
      const pt = Number(document.getElementById('qSplitPointAmount').value) || 0;
      const accSel = document.getElementById('qAccount');
      const ptSel = document.getElementById('qSplitPointAccount');
      const accName = accSel.options[accSel.selectedIndex]?.text || 'メイン口座';
      const ptName = ptSel.options[ptSel.selectedIndex]?.text || 'ポイント口座';
      const preview = document.getElementById('splitPreviewText');

      if (total <= 0 || pt <= 0) {
        preview.textContent = '合計金額と使用ポイントを入力してください';
        return;
      }
      const mainAmt = total - pt;
      if (mainAmt <= 0) {
        preview.innerHTML = `<span style="color:#d12c2c;font-weight:700;">※使用ポイントは合計 ${total.toLocaleString()}円 より少なくしてください</span>`;
        return;
      }
      preview.innerHTML = `① <b>${accName}</b>：${mainAmt.toLocaleString()}円<br>② <b>${ptName}</b>：${pt.toLocaleString()}円`;
    }

    function addMultiChip(type, selectEl) {
      const val = selectEl.value;
      selectEl.value = '';
      if (!val || selectedMulti[type].includes(val)) return;
      selectedMulti[type].push(val);
      renderMultiChips(type);
    }

    function removeMultiChip(type, val) {
      selectedMulti[type] = selectedMulti[type].filter(x => x !== val);
      renderMultiChips(type);
    }

    function renderMultiChips(type) {
      const containerId = type === 'genres' ? 'qGenreChips' : 'qMemoChips';
      document.getElementById(containerId).innerHTML = selectedMulti[type].map(v =>
        `<span class="chip">${v}<button type="button" onclick="removeMultiChip('${type}','${v.replace(/'/g, "\\'")}')">×</button></span>`
      ).join('');
    }

    function openTagModal(defaultTarget) {
      document.getElementById('modalStatus').textContent = '';
      document.getElementById('newTagName').value = '';
      document.querySelectorAll('#tagModal .target-checks input').forEach(cb => {
        cb.checked = (cb.value === defaultTarget);
      });
      renderRefTags();
      document.getElementById('tagModal').classList.remove('hidden');
    }

    function closeTagModal() {
      document.getElementById('tagModal').classList.add('hidden');
    }

    function renderRefTags() {
      const checkedTargets = Array.from(document.querySelectorAll('#tagModal .target-checks input:checked')).map(cb => cb.value);
      const listEl = document.getElementById('refTagList');
      if (checkedTargets.length === 0) {
        listEl.innerHTML = '<div style="padding:10px;color:#888;font-size:12px;">上の分類にチェックを入れると既存タグ一覧が表示されます</div>';
        return;
      }
      const primaryTarget = checkedTargets[0];
      const filtered = allTags.filter(t => t.targets.includes(primaryTarget));

      if (filtered.length === 0) {
        listEl.innerHTML = '<div style="padding:10px;color:#888;font-size:12px;">まだタグがありません</div>';
        document.getElementById('newTagSort').value = 1000;
        return;
      }

      const normalOnes = filtered.filter(t => t.visibility.indexOf('下') < 0 && t.visibility.indexOf('非表示') < 0 && t.sortNum < 9000000);
      if (normalOnes.length > 0) {
        const maxSort = Math.max(...normalOnes.map(t => t.sortNum));
        document.getElementById('newTagSort').value = maxSort + 1000;
      }

      listEl.innerHTML = filtered.map(t => {
        const numStr = t.sortNum >= 9000000 ? '-' : t.sortNum;
        const nextSuggested = t.sortNum >= 9000000 ? 1000 : t.sortNum + 100;
        return `<div class="ref-item" onclick="document.getElementById('newTagSort').value=${nextSuggested}">
          <div><span class="ref-num">${numStr}</span> <b>${t.name}</b></div>
          <div><span class="ref-badge">${t.colorName}</span> <span class="ref-badge">${t.visibility}</span></div>
        </div>`;
      }).join('');
    }

    async function submitNewTag() {
      const targets = Array.from(document.querySelectorAll('#tagModal .target-checks input:checked')).map(cb => cb.value);
      const name = document.getElementById('newTagName').value.trim();
      const sortNum = document.getElementById('newTagSort').value;
      const colorName = document.getElementById('newTagColor').value;
      const visibility = document.getElementById('newTagVisibility').value;
      const statusEl = document.getElementById('modalStatus');

      if (targets.length === 0) { statusEl.textContent = '分類を1つ以上選んでください'; return; }
      if (!name) { statusEl.textContent = 'タグ名を入力してください'; return; }

      const btn = document.getElementById('saveTagBtn');
      btn.disabled = true;
      btn.textContent = '作成＆Notionへ同期中...';
      statusEl.textContent = '';

      try {
        const res = await callGas('createAndSyncTag', { name, targets, sortNum, colorName, visibility });
        ledgerOptions = res.ledgerOptions;
        allTags = res.allTags;

        const prevMain = document.getElementById('qMainCat').value;
        const prevSub = document.getElementById('qSubCat').value;
        refreshTagDropdowns();
        document.getElementById('qMainCat').value = res.targets.includes('大項目') ? res.createdName : prevMain;
        document.getElementById('qSubCat').value = res.targets.includes('小項目') ? res.createdName : prevSub;

        if (res.targets.includes('界隈ジャンル') && !selectedMulti.genres.includes(res.createdName)) {
          selectedMulti.genres.push(res.createdName);
          renderMultiChips('genres');
        }
        if (res.targets.includes('用途・メモ') && !selectedMulti.memos.includes(res.createdName)) {
          selectedMulti.memos.push(res.createdName);
          renderMultiChips('memos');
        }
        closeTagModal();
      } catch (err) {
        statusEl.textContent = 'エラー: ' + (err.message || String(err));
      } finally {
        btn.disabled = false;
        btn.textContent = '作成して家計簿に即時反映';
      }
    }

    function onPointAccountChange() {
      const sel = document.getElementById('ptAccountSelect');
      const name = sel.options[sel.selectedIndex]?.text || 'ポイント';
      document.getElementById('ptTitle').value = `${name} 月末付与`;
    }

    function addPointChunk() {
      const inp = document.getElementById('ptItemInput');
      const val = Number(inp.value);
      if (!val || val <= 0) return;
      pointChunks.push(val);
      inp.value = '';
      renderPointChunks();
      renderPointTotal();
    }

    function removePointChunk(idx) {
      pointChunks.splice(idx, 1);
      renderPointChunks();
      renderPointTotal();
    }

    function renderPointChunks() {
      document.getElementById('ptChunksList').innerHTML = pointChunks.map((v, i) =>
        `<span class="pt-chip">+${v.toLocaleString()}円<button type="button" onclick="removePointChunk(${i})">×</button></span>`
      ).join('');
    }

    function calcPointTotal() {
      const unadded = Number(document.getElementById('ptItemInput').value) || 0;
      const sum = pointChunks.reduce((a, b) => a + b, 0);
      return sum + (unadded > 0 ? unadded : 0);
    }

    function renderPointTotal() {
      document.getElementById('ptTotalDisplay').textContent = `${calcPointTotal().toLocaleString()} 円`;
    }

    function calcNextPayDate(payDay) {
      const today = new Date();
      let y = today.getFullYear(), m = today.getMonth();
      if (today.getDate() > payDay) m += 1;
      const lastDay = new Date(y, m + 1, 0).getDate();
      let d = new Date(y, m, Math.min(payDay, lastDay));
      if (d.getDay() === 6) d.setDate(d.getDate() + 2);
      else if (d.getDay() === 0) d.setDate(d.getDate() + 1);
      return formatDate(d);
    }

    function buildBankOptions(excludeId, placeholder) {
      const others = allAccounts.filter(a => a.id !== excludeId)
        .sort((a, b) => (b.accountType === '銀行' ? 1 : 0) - (a.accountType === '銀行' ? 1 : 0));
      return `<option value="">${placeholder}</option>` + others.map(a => `<option value="${a.id}">${a.name}</option>`).join('');
    }

    function onCardChange() {
      const card = allAccounts.find(a => a.id === document.getElementById('cardSelect').value);
      const disp = document.getElementById('cardBankDisplay'), sel = document.getElementById('cardBankSelect');
      if (!card) return;
      if (card.baseBank) {
        selectedCardBankId = card.baseBank;
        disp.textContent = allAccounts.find(a => a.id === card.baseBank)?.name || '未設定';
        disp.classList.remove('hidden'); sel.classList.add('hidden');
      } else {
        selectedCardBankId = null;
        sel.innerHTML = buildBankOptions(card.id, '選択...');
        disp.classList.add('hidden'); sel.classList.remove('hidden');
      }
      document.getElementById('cardPayDate').value = card.payDay ? calcNextPayDate(card.payDay) : formatDate(new Date());
    }

    function onPayoutChange() {
      const src = allAccounts.find(a => a.id === document.getElementById('payoutSourceSelect').value);
      const disp = document.getElementById('payoutBankDisplay'), sel = document.getElementById('payoutBankSelect');
      if (!src) return;
      if (src.baseBank) {
        selectedPayoutBankId = src.baseBank;
        disp.textContent = allAccounts.find(a => a.id === src.baseBank)?.name || '未設定';
        disp.classList.remove('hidden'); sel.classList.add('hidden');
      } else {
        selectedPayoutBankId = null;
        sel.innerHTML = buildBankOptions(src.id, '選択...');
        disp.classList.add('hidden'); sel.classList.remove('hidden');
      }
      document.getElementById('payoutPayDate').value = src.payDay ? calcNextPayDate(src.payDay) : formatDate(new Date());
    }

    function setBusy(busy) {
      const btn = document.getElementById('execBtn');
      btn.disabled = busy;
      btn.classList.remove('success');
      btn.textContent = busy ? '送信中...' : getSubmitLabel(currentMode);
    }

    function showSuccess(msg) {
      const btn = document.getElementById('execBtn');
      btn.disabled = false;
      btn.classList.add('success');
      btn.textContent = msg || '✅ 完了しました！';
      document.getElementById('status').textContent = '';

      document.getElementById('qAmount').value = '';
      document.getElementById('qTitle').value = '';
      document.getElementById('qStore').value = '';
      document.getElementById('qUsePointSplit').checked = false;
      document.getElementById('qSplitPointAmount').value = '';
      togglePointSplit();
      selectedMulti = { genres: [], memos: [] };
      renderMultiChips('genres');
      renderMultiChips('memos');

      pointChunks = [];
      document.getElementById('ptItemInput').value = '';
      renderPointChunks();
      renderPointTotal();

      document.getElementById('amountInput').value = '';
      document.getElementById('feeInput').value = '';
      document.getElementById('cardAmountInput').value = '';
      document.getElementById('cardPointAmount').value = '';
      document.getElementById('payoutAmountInput').value = '';

      setTimeout(() => {
        btn.classList.remove('success');
        btn.textContent = getSubmitLabel(currentMode);
      }, 2500);
    }

    function showError(err) {
      setBusy(false);
      document.getElementById('status').textContent = 'エラー: ' + (err.message || String(err));
    }

    async function handleSubmit() {
      document.getElementById('status').textContent = '';
      try {
        if (currentMode === 'quick') {
          const usePointSplit = document.getElementById('qUsePointSplit').checked;
          const data = {
            inout: currentInout,
            date: document.getElementById('qDate').value,
            amount: document.getElementById('qAmount').value,
            accountId: document.getElementById('qAccount').value,
            title: document.getElementById('qTitle').value,
            mainCat: document.getElementById('qMainCat').value,
            subCat: document.getElementById('qSubCat').value,
            genres: selectedMulti.genres,
            memos: selectedMulti.memos,
            store: document.getElementById('qStore').value,
            taxRate: document.getElementById('qTaxRate').value,
            usePointSplit: usePointSplit,
            pointAccountId: document.getElementById('qSplitPointAccount').value,
            pointAmount: document.getElementById('qSplitPointAmount').value
          };
          if (!data.amount || Number(data.amount) <= 0) { document.getElementById('status').textContent = '金額を入力してください'; return; }
          setBusy(true);
          const r = await callGas('saveQuickTransaction', data);
          showSuccess(r.count === 2 ? '✅ 2行に分割して記録しました！' : '✅ 記録しました！');

        } else if (currentMode === 'point') {
          const total = calcPointTotal();
          if (total <= 0) { document.getElementById('status').textContent = 'ポイント数を入力してください'; return; }
          const data = {
            pointAccountId: document.getElementById('ptAccountSelect').value,
            date: document.getElementById('ptDate').value,
            title: document.getElementById('ptTitle').value,
            amount: total
          };
          setBusy(true);
          await callGas('saveMonthlyPoints', data);
          showSuccess('✅ 月末ポイントを記録しました！');

        } else if (currentMode === 'transfer') {
          const data = {
            purpose: document.getElementById('purpose').value,
            date: document.getElementById('dateInput').value,
            fromAccount: document.getElementById('fromAccount').value,
            toAccount: document.getElementById('toAccount').value,
            amount: document.getElementById('amountInput').value,
            fee: document.getElementById('feeInput').value
          };
          if (!data.amount) { document.getElementById('status').textContent = '金額を入力してください'; return; }
          setBusy(true);
          await callGas('processTransfer', data);
          showSuccess('✅ 振替を記録しました！');

        } else if (currentMode === 'card') {
          const sel = document.getElementById('cardSelect');
          const card = allAccounts.find(a => a.id === sel.value);
          const bankId = selectedCardBankId || document.getElementById('cardBankSelect').value;
          const data = {
            cardId: sel.value,
            cardName: sel.options[sel.selectedIndex]?.text || 'カード',
            bankId: bankId,
            payDate: document.getElementById('cardPayDate').value,
            amount: document.getElementById('cardAmountInput').value,
            pointAmount: document.getElementById('cardPointAmount').value,
            pointAccountId: card ? card.pointAccount : null
          };
          if (!data.cardId || !data.bankId || !data.amount) { document.getElementById('status').textContent = 'カード・口座・金額を入力してください'; return; }
          setBusy(true);
          await callGas('scheduleCardPayment', data);
          showSuccess('✅ 引落予約完了！');

        } else if (currentMode === 'payout') {
          const sel = document.getElementById('payoutSourceSelect');
          const bankId = selectedPayoutBankId || document.getElementById('payoutBankSelect').value;
          const data = {
            sourceId: sel.value,
            sourceName: sel.options[sel.selectedIndex]?.text || '報酬',
            bankId: bankId,
            payDate: document.getElementById('payoutPayDate').value,
            amount: document.getElementById('payoutAmountInput').value
          };
          if (!data.sourceId || !data.bankId || !data.amount) { document.getElementById('status').textContent = '振込元・口座・金額を入力してください'; return; }
          setBusy(true);
          await callGas('schedulePayout', data);
          showSuccess('✅ 振込予約完了！');
        }
      } catch (err) {
        showError(err);
      }
    }
